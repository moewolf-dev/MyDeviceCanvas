# Optional ESP-IDF project. Configure only when IDF_PATH is available.
cmake_minimum_required(VERSION 3.16)

if(DEFINED ENV{IDF_PATH})
  include($ENV{IDF_PATH}/tools/cmake/project.cmake)
  project(mdc_esp32_runtime)
else()
  message(STATUS "IDF_PATH unset — building host bring-up stub")
  project(mdc_esp32_idf_stub LANGUAGES C)
  add_library(mdc_idf_stub STATIC main/mdc_app_main.c)
  target_include_directories(mdc_idf_stub PUBLIC port)
  add_executable(mdc_panel_bringup_test tests/panel_bringup_test.c)
  target_link_libraries(mdc_panel_bringup_test PRIVATE mdc_idf_stub)
  enable_testing()
  add_test(NAME panel_bringup COMMAND mdc_panel_bringup_test)
endif()
